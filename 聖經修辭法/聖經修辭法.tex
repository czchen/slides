\section{簡介}

\begin{frame}{什麼是聖經修辭法？}
  聖經原文作者在寫作時，為了傳達神學信息、強化情感表達、增強記憶與說服力所使用的各類文學結構、語義技巧與修辭手法。\parvspace
  聖經修辭法的目的如下：
  \begin{itemize}
    \item 促進口傳記憶：藉由對稱、平行與韻律結構，利於古代口傳文化的吟誦、聆聽與長期記憶。
    \item 突顯神學高峰：運用交叉對稱與結構鋪陳，將最核心的神學焦點與真理置於篇章高潮。
    \item 觸動聽覺想像：透過鮮明意象與聲韻修辭，喚起宣講與聆聽時的感官共鳴與情感震撼。
    \item 彰顯救贖歷史連貫性：結合預表與救贖敘事模式，展現貫穿新舊約聖經貫通一致的神聖計畫。
  \end{itemize}
\end{frame}

\begin{frame}{為何需要認識聖經的修辭法？}
  \begin{itemize}
    \item 避免極端字面主義：正確識別象徵與比喻，防止曲解或誤讀聖經原意。
    \item 還原作者原意：辨析文學體裁與表達手法，精準掌握原作者要傳遞的核心信息。
    \item 領會文學美感與靈性力道：體會希伯來詩歌與先知預言中意象的感染力與靈性震撼。
  \end{itemize}
\end{frame}

\section{結構與篇章模式}

\begin{frame}{結構與篇章模式}
  \begin{itemize}
    \item 平行體 (Parallelism)
    \item 交叉對稱 (Chiasmus)
    \item 首尾呼應 (Inclusio)
    \item 兩極表達法 (Merism)
    \item 離合詩/藏頭詩 (Acrostic)
    \item 相對否定 (Relative Negation: Not X, But Y)
    \item 一詞七用 (Sevenfold Repetition)
  \end{itemize}
\end{frame}

\begin{frame}{平行體 (Parallelism)}
  \begin{itemize}
    \item 平行體\parencite{Parallelism}是希伯來詩歌的核心結構與修辭基石。連續詩句在意涵、句法或思想結構上相互呼應。不同於西方詩歌依賴格律與押韻，希伯來詩歌運作於「思想、意象與結構概念的押韻」。
    \item 住要功能：
      \begin{itemize}
        \item 深化與澄清：第二句詩對第一句的思想進行深化、補充、澄清或落實根基。
        \item 建構記憶結構：韻律上的對稱性，幫助古代以色列人利於公共敬拜中口誦與記憶詩篇、箴言與先知神諭。
        \item 展現多維度的真理：將呼應的詩句放在一起解讀，能揭示比單句更為豐富、細緻與全面的真理。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{主要平行體}
  \begin{itemize}
    \item 同義 (Synonymous) 平行 ：第二句使用不同詞彙重述或強化第一句的核心概念。\\
      (\bibleref{Psalms 19:1}) 諸天述說神的榮耀；穹蒼傳揚他的手段。
    \item 對照 (Antithetic) 平行：第二句與第一句形成鮮明對比，透過反襯突顯真理與智慧。\\
      (\bibleref{Proverbs 10:1}) ... 智慧之子使父親歡樂；愚昧之子叫母親擔憂。
    \item 綜合 (Synthetic) 平行：第二句不單純重複，而是補充、延伸或進一步發展第一句的思想。\\
      (\bibleref{Psalm 23:1-2}) 耶和華是我的牧者，我必不致缺乏。他使我躺臥在青草地上，領我在可安歇的水邊。
  \end{itemize}
\end{frame}

\begin{frame}{其他平行體}
  \begin{itemize}
    \item 交叉 (Chiastic) 平行：兩句或多句的語法結構與概念呈倒置對稱 (A-B // B'-A')。\\
      (\bibleref{Matthew 23:12}) 凡高抬自己的，必被降卑；凡自己謙卑的，必被升高。\\
      \begin{hebrew}
        (\biblerefhebrew{Proverbs 6:23}) ... נֵר מִצְוָה וְתוֹרָה אוֹר ...
      \end{hebrew}
    \item 階梯 (Stair-like) 平行：第二句重複第一句的部分詞彙，並層層遞進累積，直到達到最高潮。\\
      (\bibleref{Psalm 29:1-2}) 神的眾子啊！要歸給耶和華，你們要把榮耀和能力歸給耶和華。要將耶和華的名所當得的榮耀歸給他，以聖潔的妝飾敬拜耶和華 。\\
    \item 喻意 (Emblematic) 平行：一句使用具體意象或明喻，另一句說明屬靈真理。\\
      (\bibleref{Psalm 103:13}) 父親怎樣憐恤他的兒女，耶和華也怎樣憐恤敬畏他的人！
  \end{itemize}
\end{frame}

\begin{frame}{交叉對稱 (Chiasm)}
  \begin{itemize}
    \item 倒置對稱結構 (A-B // B'-A' 或 A-B-X-B'-A') \parencite{BiblicalChiasmExchange}。
    \item 核心焦點 (Pivot X)：結構中心通常是全篇的神學核心焦點或重大轉折。
    \item 交叉對稱的四個結構層級：
      \begin{itemize}
        \item 微觀交叉對稱：單句/短語層面
        \item 段落交叉對稱：獨立敘事/單篇詩歌層面
        \item 宏觀/書卷級交叉對稱：跨章節/整卷書層面
        \item 嵌套交叉對稱：交叉結構內部嵌入子交叉
      \end{itemize}
    \item 交叉平行 (Chiastic Parallelism) 也是一種交叉對稱 (Chiasm)。
  \end{itemize}
\end{frame}

\begin{frame}{微觀交叉對稱}
  \begin{itemize}
    \item 發生於單一句子或單節經文內部的詞性、句法或邏輯級倒裝。
    \item 勝過罪的終極根基 (\bibleref{IJohn 3:9})
      \begin{itemize}
        \item (A) 凡是從 \textbf{神生}的，
        \item \quad (B) 就不 \textbf{犯罪}，
        \item \quad\quad \textbf{(X) 因為神的生命存在他心裡；}
        \item \quad (B') 他也不能 \textbf{犯罪}，
        \item (A') 因為他是從 \textbf{神生} 的。
      \end{itemize}
  \end{itemize}
\end{frame}
\begin{frame}{段落交叉對稱}

\begin{frame}{首尾呼應 (Inclusio)}
  \begin{itemize}
    \item 在一個文學段落或章節的開頭與結尾使用相同的詞彙、短語或主題，劃定獨立文學單元的邊界，並強調該段落的核心主題。
    \item 創造大篇章 (\bibleref{Genesis 1:1} \textemdash\ \bibleref{Genesis 2:1})
      \begin{itemize}
        \item 首 (\bibleref{Genesis 1:1}) 起初，神創造\textbf{天地}。
        \item 尾 (\bibleref{Genesis 2:1}) \textbf{天地}萬物都造齊了。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{兩極表達法 (Merism)}
  \begin{itemize}
    \item 使用兩個對立或相對極端的詞彙，來代表包含其間的一切整體。
    \item (\bibleref{Genesis 1:1}) 起初，神創造\textbf{天地}。
  \end{itemize}
\end{frame}

\begin{frame}{離合詩 (Acrostic)}
  \begin{itemize}
    \item 依照希伯來文 22 個字母順序 (\texthebrew{א} 至 \texthebrew{ת}) 安排每句／每組開頭字母的詩歌結構。
    \item (\bibleref{Psalms 119}) 神話語（律法）的卓越、全備、神聖與永恆
    \item (\bibleref{Proverbs 31:10-31}) 歌頌敬畏神婦人的完全美德
  \end{itemize}
\end{frame}

\begin{frame}{相對否定 (Relative Negation: Not X, But Y)}
  \begin{itemize}
    \item 在\textcite{Berakhot12b13a}解釋 Not X, But Y 表示\textbf{X 退為次要，Y 升為核心首要}，而不是廢除 X。
    \item (\bibleref{Deuteronomy 5:3}) 這約不是與我們列祖立的，乃是與我們今日在這裡存活之人立的 。
    \item (\bibleref{Jeremiah 7:22-23}) ... 燔祭平安祭的事我並沒有提說 ... 我只吩咐他們這一件說：你們當聽從我的話 ...
    \item (\bibleref{John 6:27}) 不要為那必壞的食物勞力，要為那存到永生的食物勞力 ...
    \item (\bibleref{Romans 3:27}) ... 憑行為嗎？不是的，而是以信心為準則說的。
  \end{itemize}
\end{frame}

\begin{frame}{一詞七用 (Seven Code)}
  \begin{itemize}
    \item 在一個敘事或段落單元內，特定關鍵字精準重複出現 \textbf{7 次}（或 7 的倍數）。
    \item 創造大篇章 (\bibleref{Genesis 1:1-2:1}) 中，\textbf{神看是好的}出現了七次 (\bibleref{Genesis 1:4,10,12,18,21,25,31})
  \end{itemize}
\end{frame}

\section{比喻與修辭格}

\begin{frame}{本章修辭格總覽}
  \begin{itemize}
    \item 比喻 (Figures of Comparison)
      \begin{itemize}
        \item 明喻 (Simile) / 暗喻 (Metaphor) / 借喻 (Hypocatastasis)
      \end{itemize}
    \item 替代與代表 (Substitutions)
      \begin{itemize}
        \item 換喻 (Metonymy) / 提喻 (Synecdoche)
      \end{itemize}
    \item 擬人與神人同形 (Personification \& Anthropomorphism)
      \begin{itemize}
        \item 擬人 (Personification) / 擬神同形 (Anthropomorphism)
      \end{itemize}
    \item 語氣與修飾 (Emphasis \& Tone)
      \begin{itemize}
        \item 誇飾 (Hyperbole) / 反諷 (Irony)
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{比喻 (Figures of Comparison)}
  \begin{itemize}
    \item 定義：透過將本體與另一 Known 事物（喻體）連結，以描繪特徵與強烈意象。
    \item 依比較強度與顯晦程度分為三種漸進層次：
      \begin{itemize}
        \item \textbf{明喻 (Simile)}：顯式比較（使用「如」、「像」；A 如 B）
        \item \textbf{暗喻 (Metaphor)}：隱式比較（使用「是」；A 是 B）
        \item \textbf{借喻 (Hypocatastasis)}：直接替代（完全隱去 A 與比較詞，直接以 B 代替 A）
      \end{itemize}
  \end{itemize}
\end{frame}
\begin{frame}{明喻 (Simile)}
  \begin{itemize}
    \item 定義：顯式的比較，通常使用「如」、「像」、「好比」等連接詞。
    \item 特點：明確指出兩事物之間的相似之處。
    \item 經文範例與應用：
      \begin{itemize}
        \item 待補充經文與解析
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{暗喻 (Metaphor)}
  \begin{itemize}
    \item 定義：隱式的比較，直接將一事物稱為另一事物（常使用「是」）。
    \item 特點：不使用比較連接詞，將兩者的特質直接連結。
    \item 經文範例與應用：
      \begin{itemize}
        \item 待補充經文與解析
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{借喻 (Hypocatastasis)}
  \begin{itemize}
    \item 定義：直接以替代事物稱呼對象，完全不提及被比較的原對象與比較詞。
    \item 特點：比暗喻更為強烈、生動且具暗示性。
    \item 經文範例與應用：
      \begin{itemize}
        \item 待補充經文與解析
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{換喻/轉喻 (Metonymy)}
  \begin{itemize}
    \item 定義：以與原事物有密切相關的詞彙來替代原事物（如以原因代結果、以容器代盛物）。
    \item 特點：利用關聯性來簡練表達概念。
    \item 經文範例與應用：
      \begin{itemize}
        \item 待補充經文與解析
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{提喻 (Synecdoche)}
  \begin{itemize}
    \item 定義：以局部代表整體，或以整體代表局部。
    \item 特點：強調特定部分或擴大代表範圍。
    \item 經文範例與應用：
      \begin{itemize}
        \item 待補充經文與解析
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{擬人 (Personification)}
  \begin{itemize}
    \item 定義：賦予無生命之物、自然界或抽象概念人的行為、情感與特質。
    \item 特點：使敘述與詩歌更具生動性與感染力。
    \item 經文範例與應用：
      \begin{itemize}
        \item 待補充經文與解析
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{擬神/神人同形同性 (Anthropomorphism)}
  \begin{itemize}
    \item 定義：使用人類的身體器官、情感或動作來描述神的屬性與作為。
    \item 特點：幫助有限的人類理解無限的神（例如神的眼、神的手、神的後悔）。
    \item 經文範例與應用：
      \begin{itemize}
        \item 待補充經文與解析
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{誇飾 (Hyperbole)}
  \begin{itemize}
    \item 有意的誇大或過度強調，以突出強烈的感情或印象，而非提供嚴格字面數據，聽眾明白其為強調手法而非字面數字。
  \end{itemize}
\end{frame}

\begin{frame}{反諷 (Irony)}
  \begin{itemize}
    \item 字面意義與真正意圖相反，用以嘲諷、警惕或深刻揭露盲點，透過反差加深讀者印象與反思。
  \end{itemize}
\end{frame}
