\section{簡介}

\begin{frame}{什麼是聖經修辭法？}
  聖經原文作者在寫作時，為了以下目的所使用的各類文學結構、語義技巧與修辭手法：\\
  \begin{itemize}
    \item \textbf{促進口傳記憶}：藉由對稱、平行與韻律結構，利於古代口傳文化的吟誦、聆聽與長期記憶。
    \item \textbf{突顯神學高峰}：運用交叉對稱與結構鋪陳，將最核心的神學焦點與真理置於篇章高潮。
    \item \textbf{觸動聽覺想像}：透過鮮明意象與聲韻修辭，喚起宣講與聆聽時的感官共鳴與情感震撼。
  \end{itemize}
\end{frame}

\begin{frame}{為何需要認識聖經的修辭法？}
  \begin{itemize}
    \item \textbf{避免極端字面主義}：正確識別象徵與比喻，防止曲解或誤讀聖經原意。
    \item \textbf{還原作者原意}：辨析文學體裁與表達手法，精準掌握原作者要傳遞的核心信息。
    \item \textbf{領會文學美感與靈性力道}：體會希伯來詩歌與先知預言中意象的感染力與靈性震撼。
  \end{itemize}
\end{frame}

\section{結構與篇章模式}

\begin{frame}{結構與篇章模式}
  \begin{itemize}
    \item 重複 (Repetition)
    \item 平行體 (Parallelism)
    \item 交叉對稱 (Chiasmus)
    \item 首尾呼應 (Inclusio)
    \item 夾層結構 (Intercalation)
    \item 一詞七用 (Sevenfold Repetition)
    \item 數字遞進 (Numerical Cadence)
    \item 舉輕明重 (Fortiori / \texthebrew{קַל וָחֹמֶר})
    \item 同詞類比 (\texthebrew{גְּזֵרָה שָׁוָה})
  \end{itemize}
\end{frame}

\begin{frame}{重複 (Repetition)}
  聖經中透過不同形式的字詞重複，來表達強調、強烈情感、神聖屬性或確立思想脈絡\parencite{Repetition}。
  \begin{itemize}
    \item\textbf{緊接重複 (Epizeuxis)}：同一詞彙連續緊接重複，表達極度強調。\\
      (\bibleref{Isaiah 6:3}) 聖哉！聖哉！聖哉！萬軍之耶和華$\dots$
    \item\textbf{首語重複 (Anaphora)}：連續句首使用相同詞彙，增強語勢與韻律。\\
      (\bibleref{ICorinthians 13:7}) 凡事包容，凡事相信，凡事盼望，凡事忍耐。
    \item\textbf{尾語重複 (Epistrophe)}：連續句尾使用相同詞彙，強化結論與節奏。\\
      (\bibleref{IICorinthians 11:22}) 他們是希伯來人麼？我也是。他們是以色列人麼？我也是。他們是亞伯拉罕的後裔麼？我也是。
  \end{itemize}
\end{frame}

\begin{frame}{平行體 (Parallelism)}
  平行體\parencite{Parallelism}是希伯來詩歌的核心結構與修辭基石。連續詩句在意涵、句法或思想結構上相互呼應。不同於西方詩歌依賴格律與押韻，希伯來詩歌運作於「思想、意象與結構概念的押韻」。\\
  主要功能：
  \begin{itemize}
    \item \textbf{深化與澄清}：第二句詩對第一句的思想進行深化、補充、澄清或落實根基。
    \item \textbf{建構記憶結構}：韻律上的對稱性，幫助古代以色列人利於公共敬拜中口誦與記憶詩篇、箴言與先知神諭。
    \item \textbf{展現多維度的真理}：將呼應的詩句放在一起解讀，能揭示比單句更為豐富、細緻與全面的真理。
  \end{itemize}
\end{frame}

\begin{frame}{主要平行體}
  \begin{itemize}
    \item\textbf{同義平行 (Synonymous Parallelism)}：第二句使用不同詞彙重述或強化第一句的核心概念。\\
      (\bibleref{Psalms 19:1}) 諸天述說神的榮耀；穹蒼傳揚他的手段。
    \item\textbf{對照平行 (Antithetic Parallelism)}：第二句與第一句形成鮮明對比，透過反襯突顯真理與智慧。\\
      (\bibleref{Proverbs 10:1}) $\dots$ 智慧之子使父親歡樂；愚昧之子叫母親擔憂。
    \item\textbf{綜合平行 (Synthetic Parallelism)}：第二句不單純重複，而是補充、延伸或進一步發展第一句的思想。\\
      (\bibleref{Psalm 23:1-2}) 耶和華是我的牧者，我必不致缺乏。他使我躺臥在青草地上，領我在可安歇的水邊。
  \end{itemize}
\end{frame}

\begin{frame}{其他平行體}
  \begin{itemize}
    \item\textbf{交叉平行 (Chiastic Parallelism)}：兩句或多句的語法結構與概念呈倒置對稱 ($A-B-B'-A'$)。\\
      (\bibleref{Matthew 23:12}) 凡高抬自己的，必被降卑；凡自己謙卑的，必被升高。\\
      \begin{hebrew}
        (\biblerefhebrew{Proverbs 6:23}) $\dots$ נֵר מִצְוָה וְתוֹרָה אוֹר $\dots$
      \end{hebrew}
    \item\textbf{階梯平行 (Stair-like Parallelism)}：第二句重複第一句的部分詞彙，並層層遞進累積，直到達到最高潮。\\
      (\bibleref{Judges 5:27}) \uline{西西拉}在他腳前曲身仆倒，在他腳前曲身倒臥；在那裡曲身，就在那裡死亡。\\
    \item\textbf{喻意平行 (Emblematic Parallelism)}：一句使用具體意象或明喻，另一句說明屬靈真理。\\
      (\bibleref{Psalm 103:13}) 父親怎樣憐恤他的兒女，耶和華也怎樣憐恤敬畏他的人！
  \end{itemize}
\end{frame}

\begin{frame}{交叉對稱 (Chiasm)}
  交叉對稱\parencite{BiblicalChiasmExchange}是指 $A-B-B'-A'$ 或 $A-B-X-B'-A'$ 的結構，結構中心通常是全篇的神學核心焦點或重大轉折。\\
  交叉對稱的四個結構層級：
  \begin{itemize}
    \item\textbf{微觀交叉對稱 (Micro-level Chiasm)}：單句/短語層面
    \item\textbf{段落交叉對稱 (Paragraph-level Chiasm)}：獨立敘事/單篇詩歌層面
    \item\textbf{宏觀/書卷級交叉對稱 (Macro/Book-level Chiasm)}：跨章節/整卷書層面
    \item\textbf{嵌套交叉對稱 (Nested Chiasm)}：交叉結構內部嵌入子交叉
  \end{itemize}
  \textbf{交叉平行 (Chiastic Parallelism)} 也是一種\textbf{交叉對稱 (Chiasm)}。
\end{frame}

\begin{frame}{微觀交叉對稱 (Micro-level Chiasm)}
  單一句子或單節經文內部的詞性、句法或邏輯級倒裝。
  \begin{itemize}
    \item 勝過罪的終極根基 (\bibleref{IJohn 3:9})\\
      \quad ($A$) 凡是從\textbf{神生}的，\\
      \quad\quad ($B$) 就不\textbf{犯罪}，\\
      \quad\quad\quad ($X$) 因為神的生命存在他心裡；\\
      \quad\quad ($B'$) 他也不能\textbf{犯罪}，\\
      \quad ($A'$) 因為他是從\textbf{神生}的。\\
  \end{itemize}
\end{frame}

\begin{frame}{段落交叉對稱 (Paragraph-level Chiasm)}
  \begin{itemize}
    \item 巴別塔 (\bibleref{Genesis 11:1-9})\\
      \quad($A$) 全地只有一種語言，人在\uline{示拿}地住下 (\bibleref{Genesis 11:1-2})\\
      \quad\quad($B$) 他們建造城和塔 (\bibleref{Genesis 11:3-4})\\
      \quad\quad\quad($C$) 傳揚他們的名，免得分散全地 (\bibleref{Genesis 11:4})\\
      \quad\quad\quad\quad($D$) 耶和華降臨要察看城與塔 (\bibleref{Genesis 11:5})\\
      \quad\quad\quad\quad\quad($X$) 人皆同一語言，所做將無所阻擋 (\bibleref{Genesis 11:6})\\
      \quad\quad\quad\quad($D'$) 耶和華變亂語言 (\bibleref{Genesis 11:7})\\
      \quad\quad\quad($C'$) 耶和華使他們分散在全地上 (\bibleref{Genesis 11:8})\\
      \quad\quad($B'$) 他們就停工，不造那城了 (\bibleref{Genesis 11:8})\\
      \quad($A'$) 名叫\uline{巴別}，耶和華變亂全地語言並分散眾人 (\bibleref{Genesis 11:9})
  \end{itemize}
\end{frame}

\begin{frame}{宏觀/書卷級交叉對稱 (Macro/Book-level Chiasm)}
  跨越數章甚至整卷書的大型神學與敘事架構。
  \begin{itemize}
    \item 會幕與神榮耀降臨 (\bibleref{Exodus 24-40})\\
      \quad($A$) 神的榮耀降臨\uline{西奈}山頂 (\bibleref{Exodus 24:12-18})\\
      \quad\quad($B$) 會幕樣式與祭司條例 (\bibleref{Exodus 25-30})\\
      \quad\quad\quad($C$) 建造材料與技工 (\bibleref{Exodus 31:1-11})\\
      \quad\quad\quad\quad($D$) 安息日規定 (\bibleref{Exodus 31:12-17})\\
      \quad\quad\quad\quad\quad($E$) 背約，金牛犢事件 (\bibleref{Exodus 31:18-33:11})\\
      \quad\quad\quad\quad\quad\quad($X$) 神的榮耀 (\bibleref{Exodus 33:12-34:9})\\
      \quad\quad\quad\quad\quad($E'$) 重新立約 (\bibleref{Exodus 34:10-35:35})\\
      \quad\quad\quad\quad($D'$) 安息日規定 (\bibleref{Exodus 35:1-3})\\
      \quad\quad\quad($C'$) 奉獻材料與建造會幕 (\bibleref{Exodus 35:4-36:7})\\
      \quad\quad($B'$) 完成會幕與祭司聖衣 (\bibleref{Exodus 36:8-39:43})\\
      \quad($A'$) 神的榮耀充滿會幕 (\bibleref{Exodus 40})
  \end{itemize}
\end{frame}

\begin{frame}{嵌套交叉對稱 (Nested Chiasm)：啟示錄的宏觀交叉架構}
  在宏觀的大型交叉對稱結構內部，嵌入另一個獨立的子交叉對稱結構。
  \begin{itemize}
    \item 啟示錄的宏觀交叉架構\parencite{RevelationSevenSeven}：\\
      \quad($A$) 七間教會 (\bibleref{Revelation 1-3})\\
      \quad\quad($B$) 七印 (\bibleref{Revelation 4-8:1})\\
      \quad\quad\quad($C$) 七號 (\bibleref{Revelation 8:2-11:19})\\
      \quad\quad\quad\quad($X$) 七個異象 (\bibleref{Revelation 12-14})\\
      \quad\quad\quad($C'$) 七碗 (\bibleref{Revelation 15-16})\\
      \quad\quad($B'$) 七個得勝 (\bibleref{Revelation 17-20})\\
      \quad($A'$) 七個新造 (\bibleref{Revelation 21-22})
  \end{itemize}
\end{frame}

\begin{frame}{嵌套交叉對稱 (Nested Chiasm)：婦人與龍的內部嵌套}
  \begin{itemize}
    \item 啟示錄第十二章內部嵌套交叉\\
      \quad($A$) 紅龍逼迫懷孕婦人，要吞吃出生的男孩 (\bibleref{Revelation 12:1-5})\\
      \quad\quad($B$) 婦人逃到曠野，被養活 1260 天 (\bibleref{Revelation 12:6})\\
      \quad\quad\quad($C$) 天上有爭戰，米迦勒戰勝了龍 (\bibleref{Revelation 12:7-8})\\
      \quad\quad\quad\quad($X$) 撒但被摔在地上，神的國度來臨 (\bibleref{Revelation 12:9-10})\\
      \quad\quad\quad($C'$) 信徒因羔羊的血與見證的道勝過龍 (\bibleref{Revelation 12:11-12})\\
      \quad\quad($B'$) 婦人飛到曠野，被養活一載二載半載 (\bibleref{Revelation 12:13-16})\\
      \quad($A'$) 紅龍發怒，與婦人其餘兒女征戰 (\bibleref{Revelation 12:17})\\
  \end{itemize}
\end{frame}

\begin{frame}{首尾呼應 (Inclusio)}
  在一個文學段落或章節的開頭與結尾使用相同的詞彙、短語或主題，劃定獨立文學單元的邊界，並強調該段落的核心主題\parencite{Inclusio}。
  \begin{itemize}
    \item 創造大篇章 (\bibleref{Genesis 1:1;2:1})
      \begin{itemize}
        \item 首 (\bibleref{Genesis 1:1}) 起初，神創造天地。
        \item 尾 (\bibleref{Genesis 2:1}) 天地萬物都造齊了。
      \end{itemize}
    \item 讚美詩 (\bibleref{Psalms 8:1,9})
      \begin{itemize}
        \item 首 (\bibleref{Psalms 8:1}) 耶和華─我們的主啊，你的名在全地何其美！
        \item 尾 (\bibleref{Psalms 8:9}) 耶和華─我們的主啊，你的名在全地何其美！
      \end{itemize}
    \item 心靈稱頌 (\bibleref{Psalms 103:1,22})
      \begin{itemize}
        \item 首 (\bibleref{Psalms 103:1}) 我的心哪，你要稱頌耶和華！
        \item 尾 (\bibleref{Psalms 103:22}) 我的心哪，你要稱頌耶和華！
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{夾層結構 (Intercalation)}
  將一段獨立敘事或事件 ($B$) 插入另一個主敘事 ($A_1 \dots A_2$) 的中間，形成 $A_1 - B - A_2$ 的夾層結構。內外情節互相解釋並加深神學意涵。
  \begin{itemize}
    \item 咒詛無花果樹與潔淨聖殿 (\bibleref{Mark 11:12-21})\parencite{CursedTheFigTree}\\
      \quad($A_1$) 耶穌咒詛徒具葉子卻無果子的無花果樹 (\bibleref{Mark 11:12-14})\\
      \quad\quad($B$) 耶穌潔淨聖殿，斥責其淪為賊窩 (\bibleref{Mark 11:15-19})\\
      \quad($A_2$) 無花果樹連根枯乾，象徵對無實質敬拜之聖殿的審判 (\bibleref{Mark 11:20-21})
    \item 睚魯求醫與血漏婦人 (\bibleref{Mark 5:21-43})\\
      \quad($A_1$) 睚魯求耶穌醫治瀕死的女兒 (\bibleref{Mark 5:21-24})\\
      \quad\quad($B$) 患十二年血漏的婦人憑信心摸衣裳得醫治 (\bibleref{Mark 5:25-34})\\
      \quad($A_2$) 耶穌使睚魯的女兒復活，印證「不要怕，只要信」(\bibleref{Mark 5:35-43})
  \end{itemize}
\end{frame}

\begin{frame}{一詞七用 (Sevenfold Repetition)}
  在一個敘事或段落單元內，特定關鍵字精準重複出現 \textbf{7 次}（或 7 的倍數）。
  \begin{itemize}
    \item 創造大篇章 (\bibleref{Genesis 1:1-2:1}) 中，\textbf{神看是好的}出現了七次 (\bibleref{Genesis 1:4,10,12,18,21,25,31})
    \item 約翰福音中，耶穌七次宣告\textbf{我是 (\textgreek{ἐγώ εἰμι})}\parencite{IAmInJohn}：
      \begin{enumerate}
        \item 對撒瑪利亞婦人：「這和你說話的就是」(\bibleref{John 4:26})
        \item 在海面上走：「是我，不要怕」(\bibleref{John 6:20})
        \item 論不信的後果：「你們若不信我是$\dots$」(\bibleref{John 8:24})
        \item 論人子被舉起：「必知道我是$\dots$」(\bibleref{John 8:28})
        \item 宣告永恆存在：「還沒有亞伯拉罕就有了我」(\bibleref{John 8:58})
        \item 預言猶大出賣：「可以信我是$\dots$」(\bibleref{John 13:19})
        \item 客西馬尼園被捕：「我就是」(\bibleref{John 18:5-6})
      \end{enumerate}
  \end{itemize}
\end{frame}

\begin{frame}{數字遞進 (Numerical Cadence)}
  透過字數、字母數或結構的遞增排列，形成數學般的精準節奏，營造層層推進、逐步達到高潮的效果，彰顯神的秩序與恩典的擴展。
  \begin{itemize}
    \item (\bibleref{Numbers 6:24-26})\\
      \begin{hebrew}
        יְבָרֶכְךָ יְהוָה וְיִשְׁמְרֶךָ\par
        יָאֵר יְהוָה פָּנָיו אֵלֶיךָ וִיחֻנֶּךָּ\par
        יִשָּׂא יְהוָה פָּנָיו אֵלֶיךָ וְיָשֵׂם לְךָ שָׁלוֹם\par
      \end{hebrew}
    \item (\bibleref{Amos 1:3}) $\dots$ 大馬士革\textbf{三番四次}地犯罪，我必不免去他的刑罰 $\dots$
    \item (\bibleref{Proverbs 6:16}) 耶和華所恨惡的有\textbf{六樣}，連他心所憎惡的共有\textbf{七樣}：
    \item (\bibleref{Proverbs 30:18}) 我所測不透的奇妙有\textbf{三樣}，連我所不知道的共有\textbf{四樣}：
  \end{itemize}
\end{frame}

\begin{frame}{舉輕明重 (Fortiori / \texthebrew{קַל וָחֹמֶר})}
  希列七則 (Rules of Hillel)\parencite{RulesOfHillel} 之首（意為「由輕至重／由小見大」）。若在次要、微小或較低條件（輕）下尚且如此，則在更重大、首要或更高條件（重）下必然更是如此：
  \begin{itemize}
    \item 天父對兒女的眷顧 (\bibleref{Matthew 6:26})：\\
      你們看那天上的飛鳥，也不種，也不收，也不積蓄在倉裡，你們的天父尚且養活牠。你們不比飛鳥貴重得多嗎？
    \item 救贖恩典的絕對確據 (\bibleref{Romans 8:32})：\\
      神既不愛惜自己的兒子，為我們眾人捨了，豈不也把萬物和他一同白白地賜給我們嗎？
  \end{itemize}
\end{frame}

\begin{frame}{同詞類比 (\texthebrew{גְּזֵרָה שָׁוָה})}
  希列七則 (Rules of Hillel)\parencite{RulesOfHillel} 之一（意為「相同的裁決/相等類比」）。當兩處不同的經文包含相同的關鍵字詞時，透過該關鍵字將兩處經文連結，互相補充與對照詮釋：
  \begin{itemize}
    \item 一切律法和先知的總綱 (\bibleref{Leviticus 19:18; Deuteronomy 6:5})：\\
      關鍵字：\texthebrew{וְאָהַבְתָּ}
    \item 神的慈悲和受苦的僕人 (\bibleref{Exodus 34:6-7; Isaiah 53})\parencite{Forgiveness}：\\
      關鍵字：\texthebrew{נָשָׂא}, \texthebrew{עָוֹן}, \texthebrew{פֶּשַׁע}, \texthebrew{חַטָּאָת}
  \end{itemize}
\end{frame}

\section{比喻與修辭格}

\begin{frame}{比喻與修辭格}
  \begin{itemize}
    \item 比喻 (Figures of Comparison)
    \item 替代與代表 (Substitutions)
    \item 委婉語 (Euphemism)
    \item 兩極表達法 (Merism)
    \item 擬人與神人同形 (Personification \& Anthropomorphism)
    \item 強調與語氣 (Emphasis \& Tone)
    \item 弔詭 (Paradox)
    \item 相對否定 (Relative Negation)
    \item 修辭問句與設問 (Rhetorical Questions \& Hypophora)
    \item 省略與呼告 (Ellipsis and Apostrophe)
    \item 多連詞 (Polysyndeton)
  \end{itemize}
\end{frame}

\begin{frame}{比喻 (Figures of Comparison)}
  透過將本體與另一已知事物（喻體）連結，以描繪特徵與強烈意象。
  \begin{itemize}
    \item\textbf{明喻 (Simile)}：顯式的比較\\
      (\bibleref{Psalm 42:1}) 神啊，我的心切慕你，如鹿切慕溪水。
    \item\textbf{暗喻 (Metaphor)}：隱式的比較\\
      (\bibleref{Matthew 5:14}) 你們是世上的光。$\dots$
    \item\textbf{借喻 (Hypocatastasis)}：直接以替代事物稱呼對象，完全不提及被比較的原對象與比較詞。\\
      (\bibleref{Luke 13:32}) 耶穌說：你們去告訴那個狐狸說： $\dots$
  \end{itemize}
\end{frame}

\begin{frame}{替代與代表 (Substitutions)}
  透過與原對象的關聯性或包含關係，使用另一個詞彙來替代，以達到簡潔或強調的效果。
  \begin{itemize}
    \item\textbf{換喻/轉喻 (Metonymy)}：以密切相關的事物替代。\\
      拉撒路與財主：(\bibleref{Luke 16:29}) \uline{亞伯拉罕}說：他們有\uline{摩西}和先知的話可以聽從。
    \item\textbf{提喻 (Synecdoche)}：以部分代表整體，或以整體代表部分。\\
      (\bibleref{Romans 3:20}) 所以凡有血氣的，沒有一個因行律法能在神面前稱義，因為律法本是叫人知罪。
  \end{itemize}
\end{frame}

\begin{frame}{委婉語 (Euphemism)}
  透過溫和或較不直接的詞彙，來替代生硬或禁忌的事物。
  \begin{itemize}
    \item (\bibleref{John 11:11}) $\dots$ 我們的朋友\uline{拉撒路}睡了，我去叫醒他。
    \item (\bibleref{ISamuel 24:3}) 到了路旁的羊圈，在那裡有洞，\uline{掃羅}進去大解（\texthebrew{לְהָסֵךְ אֶת־רַגְלָיו}/遮掩他的腳）$\dots$
  \end{itemize}
\end{frame}

\begin{frame}{兩極表達法 (Merism)}
  使用兩個對立或相對極端的詞彙，來代表包含其間的一切整體。
  \begin{itemize}
    \item (\bibleref{Genesis 1:1}) 起初，神創造天地
    \item (\bibleref{Psalms 121:8}) 你出你入，耶和華要保護你，從今時直到永遠。
    \item (\bibleref{Revelation 22:13}) 我是阿拉法，我是俄梅戛；我是首先的，我是末後的；我是初，我是終。
    \item (\bibleref{Matthew 23:35}) $\dots$ 從義人\uline{亞伯}的血起，直到你們在殿和壇中間所殺的\uline{巴拉加}的兒子\uline{撒迦利亞}的血為止。
  \end{itemize}
\end{frame}

\begin{frame}{擬人與神人同形 (Personification \& Anthropomorphism)}
  藉由借用人類的特徵、情感或活動，投射於非人事物或神身上，使抽象真理具象化。
  \begin{itemize}
    \item\textbf{擬人 (Personification)}：賦予無生命之物、自然界或抽象概念人的行為、情感與生命特質，增強感染力。\\
      (\bibleref{Isaiah 55:12}) $\dots$ 大山小山必在你們面前發聲歌唱；田野的樹木也都拍掌。\\
    \item\textbf{神人同形 (Anthropomorphism)}：使用人類的身體器官、外貌、情感（神人同情）或行動來描述神的屬性與作為，幫助有限的人類體會無限的神。\\
      (\bibleref{Isaiah 59:1}) 耶和華的膀臂並非縮短，不能拯救，耳朵並非發沉，不能聽見，\\
  \end{itemize}
\end{frame}

\begin{frame}{強調與語氣 (Emphasis \& Tone)}
  透過超出字面常規的表達強度或語氣反差，突破慣性思維以震撼人心並深化反思。
  \begin{itemize}
    \item\textbf{誇飾 (Hyperbole)}：有意的誇大或過度強調，以突出強烈的感情或印象，而非提供嚴格字面數據。\\
      (\bibleref{Matthew 23:24})  你們這瞎眼領路的，蠓蟲你們就濾出來，駱駝你們倒吞下去。
    \item\textbf{反諷 (Irony)}：字面意義與真正意圖相反，透過反差嘲諷偽善、警惕盲點並引發深刻反思。\\
      (\bibleref{IKings 18:27}) 到了正午，\uline{以利亞}嬉笑他們，說：大聲求告罷！因為他是神，他或默想，或走到一邊，或行路，或睡覺，你們當叫醒他。
    \item\textbf{階梯遞進 (Climax)}：語意、情緒或論點由淺入深、層層推進，逐步累積張力以達至最高潮。\\
      (\bibleref{Romans 5:3-5}) $\dots$ 因為知道患難生忍耐，忍耐生老練，老練生盼望；盼望不至於羞恥 $\dots$
  \end{itemize}
\end{frame}

\begin{frame}{弔詭 (Paradox)}
  提出表面看似矛盾或荒謬的陳述，深入思考後卻揭示更深層的屬靈真理。
  \begin{itemize}
    \item (\bibleref{IICorinthians 6:10}) 似乎憂愁，卻是常常快樂的；似乎貧窮，卻是叫許多人富足的；似乎一無所有，卻是樣樣都有的。
  \end{itemize}
\end{frame}

\begin{frame}{相對否定 (Relative Negation)}
  在語法形式上使用\textbf{絕對否定}，但其真實意圖並非完全廢除、否定該事物，而是用來表達\textbf{優先順序}、\textbf{程度差異}或\textbf{條件限制}\parencite{Berakhot12b13a}。
  \begin{itemize}
    \item (\bibleref{Deuteronomy 5:3}) 這約不是與我們列祖立的，乃是與我們今日在這裡存活之人立的。
    \item (\bibleref{Jeremiah 7:22-23}) $\dots$ 燔祭平安祭的事我並沒有提說 $\dots$ 我只吩咐他們這一件說：你們當聽從我的話 $\dots$
    \item (\bibleref{Mark 9:37}) $\dots$ 凡接待我的，不是接待我，乃是接待那差我來的。
    \item (\bibleref{John 6:27}) 不要為那必壞的食物勞力，要為那存到永生的食物勞力 $\dots$
  \end{itemize}
\end{frame}

\begin{frame}{修辭問句與設問 (Rhetorical Questions \& Hypophora)}
  透過問句的形式，而非平鋪直敘的陳述，來引導聽眾的思維方向並增強論證力道。
  \begin{itemize}
    \item\textbf{設問 (Hypophora)}：講員或作者自己提出問題，然後立刻親自給出明確的回答。\\
      (\bibleref{Romans 6:1-2}) 這樣，怎麼說呢？我們可以仍在罪中、叫恩典顯多嗎？斷乎不可！我們在罪上死了的人豈可仍在罪中活著呢？
    \item\textbf{反問 (Rhetorical Question)}：提出問題但不期望對方回答，因為答案已不言自明。\\
      (\bibleref{Romans 8:31}) 既是這樣，還有甚麼說的呢？神若幫助我們，誰能敵擋我們呢？
  \end{itemize}
\end{frame}

\begin{frame}{省略與呼告 (Ellipsis and Apostrophe)}
  透過省去文法字詞或突然轉換發話對象，營造緊湊語氣並激發強烈的情感張力。
  \begin{itemize}
    \item\textbf{省略 (Ellipsis)}：省去文法上必需但上下文可推知的字詞，使語意更為精煉緊湊，或突顯強烈激動的情感。\\
      (\bibleref{Psalm 6:3}) 我的心也大大戰慄，耶和華啊！要等到幾時呢？
    \item\textbf{呼告 (Apostrophe)}：講員或作者突然中斷正常語序，轉向呼喚不在現場的人、抽象概念或無生命事物。\\
      (\bibleref{ICorinthians 15:54-56}) 這必朽壞的既變成不朽壞的，這必死的既變成不死的，那時經上所記死被得勝吞滅的話就應驗了。死阿！你得勝的權勢在那裡？死阿！你的毒鉤在那裡？死的毒鉤就是罪，罪的權勢就是律法 。
  \end{itemize}
\end{frame}

\begin{frame}{多連詞 (Polysyndeton)}
  在詞彙或子句間刻意連續使用連接詞（如希伯來文 \texthebrew{וְ}、希臘文 \textgreek{καὶ}）。\\
  中文常以標點取代而淡化原文節奏，依語境具雙重修辭效果：
  \begin{itemize}
    \item\textbf{名詞列舉：放慢節奏、逐項強調（司法盤點與層層堆疊）}
      \begin{itemize}
        \item (\bibleref{Joshua 7:24}) $\dots$ 把\textbf{\uline{謝拉}的曾孫\uline{亞干}和那銀子、那件衣服、那條金子，並\uline{亞干}的兒女、牛、驢、羊、帳棚，以及他所有的}，都帶到\uline{亞割谷}去。
        \item (\bibleref{Revelation 5:12}) 大聲說：曾被殺的羔羊是配得\textbf{權柄、豐富、智慧、能力、尊貴、榮耀、頌讚}的。
      \end{itemize}
    \item\textbf{動詞串接：加快節奏、急促緊湊（連續動作一氣呵成）}
      \begin{itemize}
        \item (\bibleref{Genesis 25:34}) 於是雅各將餅和紅豆湯給了以掃，以掃\textbf{吃了喝了，便起來走了(\texthebrew{וַיֹּאכַל וַיֵּשְׁתְּ וַיָּקָם וַיֵּלַךְ}})。這就是以掃輕看了他長子的名分。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{問題 (\bibleref{Isaiah 55:8-9})}
  耶和華說：“我的意念不是你們的意念，\\
  你們的道路也不是我的道路。\\
  天怎樣高過地，\\
  我的道路也怎樣高過你們的道路，\\
  我的意念也怎樣高過你們的意念。\\

\begin{frame}{參考解答 (\bibleref{Isaiah 55:8-9})}
  \begin{itemize}
    \item\textbf{嵌套交叉對稱}：
      \begin{itemize}
        \item ($A$) 意念 ($B$) 道路 ($X$) 天高過地 ($B'$) 道路 ($A'$) 意念
        \item ($a$) 我的意念 ($b$) 你們的意念 ($b'$) 你們的道路 ($a'$) 我的道路
      \end{itemize}
    \item\textbf{同義平行}：
      \begin{itemize}
        \item 我的意念不是你們的意念，你們的道路也不是我的道路。
        \item 我的道路也怎樣高過你們的道路，我的意念也怎樣高過你們的意念。
      \end{itemize}
    \item\textbf{喻意平行/明喻}：天怎樣高過地，我的道路也怎樣高過你們的道路$\dots$
  \end{itemize}
\end{frame}

\begin{frame}{問題 (\bibleref{Amos 5:21-24})}
  我厭惡你們的節期，也不喜悅你們的嚴肅會。\\
  你們雖然向我獻燔祭和素祭，我卻不悅納，也不顧你們用肥畜獻的平安祭；\\
  要使你們歌唱的聲音遠離我，因為我不聽你們彈琴的響聲。\\
  惟願公平如大水滾滾，使公義如江河滔滔。

\begin{frame}{參考解答 (\bibleref{Amos 5:21-24})}
  \begin{itemize}
    \item\textbf{相對否定}：否定徒具外表之祭祀與節期，要求實踐真實公平與公義。
    \item\textbf{同義平行}：
      \begin{itemize}
        \item 惟願公平如大水滾滾，使公義如江河滔滔。
      \end{itemize}
    \item\textbf{明喻}：
      \begin{itemize}
        \item 如大水滾滾
        \item 如江河滔滔
      \end{itemize}
    \item\textbf{反諷}：神宣告厭惡自己所設定之宗教節期，反襯缺乏公義之虛偽。
  \end{itemize}
\end{frame}

    \item\textbf{反問}：

        \item 世人算甚麼$\dots$？
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{問題 (\bibleref{Ecclesiastes 7:28})}
  我的心仍在尋覓，卻還找不到，\\
  就是在一千個男人中，我找到一個正直人，\\
  但在所有的女人中，連一個也找不到。
\end{frame}

\begin{frame}{參考解答 (\bibleref{Ecclesiastes 7:28})}
  \begin{itemize}
    \item\textbf{同義平行}
    \item\textbf{兩極表達法}：以「男」與「女」涵蓋人類全體，哀嘆墮落世間尋覓真正正直人之罕見
    \item\textbf{誇飾}：以「千分之一」對比「零」之極端比例表達尋求完全正直人之艱難
  \end{itemize}
\end{frame}

\begin{frame}{問題 (\bibleref{John 6:35;8:12;10:7,11;11:25;14:6;15:1})}
  \begin{enumerate}
    \item 「我就是生命的糧。到我這裡來的，必定不餓；信我的，永遠不渴。」(\bibleref{John 6:35})
    \item $\dots$「我是世界的光。跟從我的，就不在黑暗裡走，必要得著生命的光。」(\bibleref{John 8:12})
    \item 「我實實在在(\textgreek{ἀμὴν ἀμὴν})地告訴你們，我就是羊的門。」(\bibleref{John 10:7})
    \item $\dots$「我是好牧人；好牧人為羊捨命。」(\bibleref{John 10:11})
    \item $\dots$「我就是復活和生命；信我的人，雖然死了，也要活著。」(\bibleref{John 11:25})
    \item 「我就是道路、真理、生命；若不藉著我，沒有人能到父那裡去。」(\bibleref{John 14:6})
    \item 「我是真葡萄樹，我父是栽培的人。」(\bibleref{John 15:1})
  \end{enumerate}
\end{frame}

\begin{frame}{參考解答 (\bibleref{John 6:35;8:12;10:7,11;11:25;14:6;15:1})}
  \begin{itemize}
    \item\textbf{暗喻}：耶穌以具體事物比喻自己的本質與救贖功能，本體與喻體直接結合。
    \item\textbf{重複/首語重複}：七次宣告「我是$\dots$」，重複相同的句型強化神聖身分與救恩主題。
    \item\textbf{一詞七用}：約翰福音中精準出現 7 次帶述語的「我是」宣告（與 7 次絕對「我是」宣告呼應），以完全數結構彰顯神性啟示的完全。
    \item\textbf{同義平行}：
      \begin{itemize}
        \item 到我這裡來的，必定不餓；信我的，永遠不渴。
      \end{itemize}
    \item\textbf{對照平行}：
      \begin{itemize}
        \item 就不在黑暗裡走，必要得著生命的光。
        \item 雖然死了，也要活著。
      \end{itemize}
    \item\textbf{緊接重複}：(\bibleref{John 10:7}) 我實實在在(\textgreek{ἀμὴν ἀμὴν})地告訴你們 $\dots$
    \item\textbf{兩極表達法}：(\bibleref{John 6:35})「餓」與「渴」以飲食兩極代表人類一切屬靈缺乏。
  \end{itemize}
\end{frame}

\begin{frame}{問題 (\bibleref{Romans 10:8-10})}
  然而那出於信心的義還說甚麼呢？\\
  “這話與你相近，在你口裡，也在你心裡。”\\
  這話就是我們所傳信心的信息。你若口裡認耶穌為主，心裡信神使他從死人中復活，就必得救； 因為心裡相信就必稱義，用口承認就必得救。\\
\end{frame}

\begin{frame}{參考解答 (\bibleref{Romans 10:8-10})}
  \begin{itemize}
    \item\textbf{設問}：然而那出於信心的義還說甚麼呢？
    \item\textbf{交叉對稱/交叉平行}：
      \begin{itemize}
        \item ($A$) 口裡 ($B$) 心裡 ($B'$) 心裡相信稱義 ($A'$) 用口承認得救
      \end{itemize}
    \item\textbf{同詞類比} (\texthebrew{גְּזֵרָה שָׁוָה})：藉\bibleref{Deuteronomy 30:14}的關鍵字「口」與「心」，類比推導出信徒「心信稱義、口認得救」
    \item\textbf{綜合平行}：
      \begin{itemize}
        \item 心裡相信就必稱義，用口承認就必得救。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{問題 (\bibleref{Romans 8:31-32,35-39})}
  既是這樣，還有甚麼說的呢？神若幫助我們，誰能敵擋我們呢？神既不愛惜自己的兒子，為我們眾人捨了，豈不也把萬物和他一同白白地賜給我們嗎？\\
  $\dots$\\
  誰能使我們與基督的愛隔絕呢？難道是患難嗎？是困苦嗎？是逼迫嗎？是飢餓嗎？是赤身露體嗎？是危險嗎？是刀劍嗎？如經上所記：“我們為你的緣故終日被殺；人看我們如將宰的羊。”然而，靠著愛我們的主，在這一切的事上已經得勝有餘了。因為我深信無論是死，是生，是天使，是掌權的，是有能的，是現在的事，是將來的事，是高處的，是低處的，是別的受造之物，都不能叫我們與神的愛隔絕；這愛是在我們的主基督耶穌裡的。
\end{frame}

\begin{frame}{參考解答 (\bibleref{Romans 8:31-32,35-39})}
  \begin{itemize}
    \item\textbf{設問與修辭問句}：
      \begin{itemize}
        \item 「還有甚麼說的呢？」「誰能敵擋我們呢？」
        \item 「誰能控告神所揀選的人呢？」「誰能使我們與基督的愛隔絕呢？」
      \end{itemize}
    \item\textbf{舉輕明重} (\texthebrew{קַל וָחֹמֶר})：神既不愛惜自己的兒子為我們捨了，豈不也把萬物白白賜給我們
    \item\textbf{兩極表達法}：死/生、天使/掌權、現在/將來、高處/低處（涵蓋宇宙萬有與時空）
    \item\textbf{首語重複}：是患難嗎？是困苦嗎？$\dots$；無論是死，是生，是天使$\dots$
  \end{itemize}
\end{frame}

\begin{frame}{問題 (\bibleref{ICorinthians 13:1-3})}
  我若能說萬人的方言，並天使的話語，卻沒有愛，我就成了鳴的鑼，響的鈸一般。我若有先知講道之能，也明白各樣的奧祕，各樣的知識，而且有全備的信，叫我能夠移山，卻沒有愛，我就算不得甚麼。我若將所有的賙濟窮人，又捨己身叫人焚燒，卻沒有愛，仍然與我無益。

\begin{frame}{參考解答 (\bibleref{ICorinthians 13:1-3})}
  \begin{itemize}
    \item\textbf{首語重複}：連續三節以「我若$\dots$卻沒有愛」句型開展
    \item\textbf{誇飾}：說萬人方言與天使話語、明白各樣奧祕知識、全備的信能夠移山、捨己身叫人焚燒
    \item\textbf{明喻}：我就成了鳴的鑼，響的鈸一般
    \item\textbf{階梯遞進}：言語恩賜 $\to$ 屬靈知識與信心 $\to$ 終極自我犧牲
  \end{itemize}
\end{frame}

\section{聲音、詩歌與風格修辭}

\begin{frame}{聲音、詩歌與風格修辭}
  \begin{itemize}
    \item 頭韻與尾韻 (Alliteration \& Rhyme)
    \item 諧音雙關 (Paronomasia / Wordplay)
    \item 疊句與複歌 (Refrain)
    \item 離合詩/藏頭詩 (Acrostic)
  \end{itemize}
\end{frame}

\begin{frame}{頭韻與尾韻 (Alliteration \& Rhyme)}
  在連續或相近的詞彙中使用相同字首子音（頭韻）或詞尾尾音（尾韻），增加朗讀時的音律節奏、莊嚴感與口傳記憶點。
  \begin{itemize}
    \item
      \begin{hebrew}
        (\biblerefhebrew{Deuteronomy 6:5}) וְאָהַבְתָּ אֵת יְהוָה אֱלֹהֶיךָ בְּכָל־לְבָבְךָ וּבְכָל־נַפְשְׁךָ וּבְכָל־מְאֹדֶךָ׃
      \end{hebrew}
    \item
      \begin{hebrew}
        (\biblerefhebrew{Psalms 122:6}) שַׁאֲלוּ שְׁלוֹם יְרוּשָׁלִָם יִשְׁלָיוּ אֹהֲבָיִךְ׃
      \end{hebrew}
  \end{itemize}
\end{frame}

\begin{frame}{諧音雙關 (Paronomasia / Wordplay)}
  利用字詞發音相似或相同但意義不同的特性，達到雙關、諷喻或加深聽覺印象的效果，常見於希伯來先知書的宣告與詩歌：
  \begin{itemize}
    \item (\bibleref{Jeremiah 1:11-12}) $\dots$「我看見一根杏樹枝(\texthebrew{שָׁקֵד}) $\dots$ 因為我留意(\texthebrew{שֹׁקֵד})保守我的話，使得成就。」
    \item (\bibleref{Isaiah 5:7}) $\dots$ 他指望的是公平(\texthebrew{מִשְׁפָּט})，誰知倒有暴虐(\texthebrew{מִשְׂפָּח})；指望的是公義(\texthebrew{צְדָקָה})，誰知倒有冤聲(\texthebrew{צְעָקָה})。
  \end{itemize}
\end{frame}

\begin{frame}{疊句與複歌 (Refrain)}
  在詩篇或哀歌中，於各段落結尾或特定間隔規律重複出現的詩句。具有劃分詩節結構、貫穿核心主題並引導會眾啟應與情感共鳴的功能。
  \begin{itemize}
    \item\textbf{禮儀啟應} (\bibleref{Psalms 136})：全篇 26 節每節句末皆重複「因他的慈愛永遠長存」，引導會眾齊聲回應宣告神永恆的慈愛。
    \item\textbf{結構分段} (\bibleref{Psalms 42:5,11; 43:5})：三次重複「我的心哪，你為何憂悶？$\dots$應當仰望神」，劃分哀歌詩節並展現心靈轉向信靠的掙扎與昇華。\parencite{Psalm4243}
    \item\textbf{輓歌哀悼} (\bibleref{IISamuel 1:19,25,27})：\uline{大衛}哀悼\uline{掃羅}與\uline{約拿單}，三次反覆哀嘆「英雄何竟仆倒！」作為全篇輓歌的結構骨幹與情感高潮。
  \end{itemize}
\end{frame}

\begin{frame}{離合詩 (Acrostic)}
  依照希伯來文 22 個字母順序 (\texthebrew{א} 至 \texthebrew{ת}) 安排每句／每組開頭字母的詩歌結構。
  \begin{itemize}
    \item \bibleref{Psalms 119}：神話語（律法）的卓越、全備、神聖與永恆。
    \item \bibleref{Proverbs 31:10-31}：歌頌敬畏神婦人的完全美德。
    \item \bibleref{Lamentations 1-4}：以字母結構承載對耶路撒冷傾覆全然、極致的哀慟（第 1、2、4 章單字母離合，第 3 章三重離合）。
  \end{itemize}
\end{frame}

\section{結論}

\begin{frame}{為何需要認識聖經的修辭法？}
  \begin{itemize}
    \item \textbf{避免極端字面主義}：正確識別象徵與比喻，防止曲解或誤讀聖經原意。
    \item \textbf{還原作者原意}：辨析文學體裁與表達手法，精準掌握原作者要傳遞的核心信息。
    \item \textbf{領會文學美感與靈性力道}：體會希伯來詩歌與先知預言中意象的感染力與靈性震撼。
  \end{itemize}
\end{frame}
